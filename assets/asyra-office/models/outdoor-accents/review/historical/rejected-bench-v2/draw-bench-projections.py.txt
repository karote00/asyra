BENCH = SPEC['modules'][0]
PARTS = {p['partId']: p for p in BENCH['components']}
SHIFT = np.array(BENCH['constructionToAssetTranslation'])
STYLE = BENCH['referenceReview']

def add(kind, centre, group, **kw):
    PRIMITIVES.append(dict(kind=kind, centre=np.array(centre)+SHIFT, group=group, **kw))


def sphere(centre, radius, group):
    add('sphere', centre, group, radius=radius)


def cylinder(a, b, radius, group):
    a, b = np.array(a), np.array(b)
    add('cylinder', (a+b)/2, group, radius=radius,
        axis=(b-a)/np.linalg.norm(b-a), half=np.linalg.norm(b-a)/2)


def rounded_box(centre, size, bevel, group, rotation=np.eye(3)):
    centre, core = np.array(centre), np.array(size)/2-bevel
    if group < 7:
        BOARD_FRAMES[group] = (centre+SHIFT, np.array(size)/2, rotation, bevel)
    for axis in range(3):
        half = core.copy(); half[axis] += bevel
        add('box', centre, group, half=half, rotation=rotation)
    for axis in range(3):
        other = [i for i in range(3) if i != axis]
        for a in (-1, 1):
            for b in (-1, 1):
                p = np.zeros(3); p[other] = core[other]*[a,b]
                start, end = p.copy(), p.copy()
                start[axis], end[axis] = -core[axis], core[axis]
                cylinder(centre+rotation@start, centre+rotation@end, bevel, group)
    for x in (-1,1):
        for y in (-1,1):
            for z in (-1,1):
                sphere(centre+rotation@(core*[x,y,z]), bevel, group)

    p = PARTS['arms']; control = np.array(p['controlPointsYZ']); x = p['xValues'][side]

        path.append(np.array([x,y,z])); sphere(path[-1],p['radius'],group)
    for a,b in zip(path,path[1:]): cylinder(a,b,p['radius'],group)

def extent(p, direction):
    if p['kind'] == 'sphere': return p['radius']
    if p['kind'] == 'box': return np.abs(direction@p['rotation'])@p['half']
    cosine = np.dot(direction,p['axis'])
    return abs(cosine)*p['half']+p['radius']*math.sqrt(max(0,1-cosine*cosine))


bounds = np.array([[min(p['centre'][i]-extent(p,np.eye(3)[i]) for p in PRIMITIVES)
                    for i in range(3)],
                   [max(p['centre'][i]+extent(p,np.eye(3)[i]) for p in PRIMITIVES)
                    for i in range(3)]])
# Analytic primitive envelopes are conservative at rounded-box corners, but
# principal extrema are attained on flat face centres for this construction.
assert np.allclose(bounds[0], BENCH['defaultBounds']['min'], atol=1e-9)
assert np.allclose(bounds[1], BENCH['defaultBounds']['max'], atol=1e-9)


def rgb(s): return [int(s[i:i+2],16) for i in (1,3,5)]


def draw_view(name, camera, right, origin, rect):
    eye = np.array(camera,dtype=float); eye /= np.linalg.norm(eye)
    right = np.array(right,dtype=float); right /= np.linalg.norm(right)
    up = np.cross(eye,right); direction = -eye
    x0,y0,x1,y1 = rect
    horizontal = [np.dot(p['centre'],right)+sign*extent(p,right) for p in PRIMITIVES for sign in (-1,1)]
    vertical = [np.dot(p['centre'],up)+sign*extent(p,up) for p in PRIMITIVES for sign in (-1,1)]
    if name == 'THREE-QUARTER':
        origin = ((x0+x1)/2-SCALE*(min(horizontal)+max(horizontal))/2,
                  (y0+y1)/2+SCALE*(min(vertical)+max(vertical))/2)
    projected = [origin[0]+SCALE*min(horizontal),origin[1]-SCALE*max(vertical),
                 origin[0]+SCALE*max(horizontal),origin[1]-SCALE*min(vertical)]
    assert projected[0]>x0+2 and projected[1]>y0+2 and projected[2]<x1-2 and projected[3]<y1-2, (name,projected,rect)
    shape = ((y1-y0)*SS,(x1-x0)*SS)
    depth = np.full(shape,np.inf)
    owners = np.full(shape,-1,dtype=np.int16)
    faces = np.zeros(shape,dtype=np.int8)
    for p in PRIMITIVES:
        cx = origin[0]+SCALE*np.dot(p['centre'],right)
        cy = origin[1]-SCALE*np.dot(p['centre'],up)
        ex,ey = SCALE*extent(p,right),SCALE*extent(p,up)
        left=max(x0*SS,int(math.floor((cx-ex)*SS))-1)
        top=max(y0*SS,int(math.floor((cy-ey)*SS))-1)
        rr=min(x1*SS,int(math.ceil((cx+ex)*SS))+1)
        bb=min(y1*SS,int(math.ceil((cy+ey)*SS))+1)
        if rr<=left or bb<=top: continue
        ys,xs = np.mgrid[top:bb,left:rr]
        rays = (((xs+.5)/SS-origin[0])/SCALE)[...,None]*right
        rays = rays+((origin[1]-(ys+.5)/SS)/SCALE)[...,None]*up+eye*10
        o = rays-p['centre']; hit=np.full(xs.shape,np.inf)
        face=np.zeros(xs.shape,dtype=np.int8)
        if p['kind']=='sphere':
            b=o@direction; c=np.sum(o*o,axis=2)-p['radius']**2
            disc=b*b-c
            hit=np.where(disc>=0,-b-np.sqrt(np.maximum(disc,0)),np.inf)
        elif p['kind']=='box':
            local=o@p['rotation']; d=direction@p['rotation']
            near=np.full(xs.shape,-np.inf); far=np.full(xs.shape,np.inf)
            valid=np.ones(xs.shape,dtype=bool)
            for a in range(3):
                if abs(d[a])<1e-12:
                    valid &= np.abs(local[...,a])<=p['half'][a]
                    continue
                t1=(-p['half'][a]-local[...,a])/d[a]
                t2=(p['half'][a]-local[...,a])/d[a]
                entry=np.minimum(t1,t2)
                face[entry>near]=a
                near=np.maximum(near,entry); far=np.minimum(far,np.maximum(t1,t2))
            hit=np.where(valid&(near<=far)&(near>=0),near,np.inf)
        else:
            axis=p['axis']; axial=o@axis; da=np.dot(direction,axis)
            perpendicular=o-axial[...,None]*axis
            dp=direction-da*axis
            a=np.dot(dp,dp); b=perpendicular@dp
            c=np.sum(perpendicular*perpendicular,axis=2)-p['radius']**2
            if a>1e-12:
                disc=b*b-a*c
                t=(-b-np.sqrt(np.maximum(disc,0)))/a
                hit=np.where((disc>=0)&(np.abs(axial+t*da)<=p['half']),t,np.inf)
            if abs(da)>1e-12:
                for sign in (-1,1):
                    t=(sign*p['half']-axial)/da
                    radial=perpendicular+t[...,None]*dp
                    valid=(np.sum(radial*radial,axis=2)<=p['radius']**2)&(t<hit)
                    hit[valid]=t[valid]
        sl=np.s_[top-y0*SS:bb-y0*SS,left-x0*SS:rr-x0*SS]
        visible=(hit>=0)&(hit<depth[sl]-1e-10)
        depth[sl][visible]=hit[visible]; owners[sl][visible]=p['group']
        faces[sl][visible]=face[visible] if p['kind']=='box' else 1
    pixels=np.full((*shape,3),255,dtype=np.uint8)
    for group in range(10):
        mask=owners==group
        if group>=7:
            pixels[mask]=rgb(PALETTE['metal'])
            continue
        yy,xx=np.where(mask)
        if not len(xx): continue
        h=((xx+x0*SS+.5)/SS-origin[0])/SCALE
        v=(origin[1]-(yy+y0*SS+.5)/SS)/SCALE
        points=h[:,None]*right+v[:,None]*up+(10-depth[mask])[:,None]*eye
        centre,half,board_rotation,bevel=BOARD_FRAMES[group]
        local=(points-centre)@board_rotation
        # Choose the nearest face for the documented local material mapping.
        # The same colour recipe is used on every orientation; no lighting.
        normals=local-np.clip(local,-half+bevel,half-bevel)
        axes=np.argmax(np.abs(normals),axis=1)
        positive=local+half
        longitudinal=positive[:,0].copy()
        cross=np.where(axes==2,positive[:,1],positive[:,2])
        longitudinal[axes==0]=positive[axes==0,1]
        f=.5+.5*np.sin(2*np.pi*(cross/.004+.1*np.sin(2*np.pi*longitudinal/.18)))
        base=np.array(rgb(PALETTE['wood']))/255
        grain=np.array(rgb(PALETTE['wood-grain']))/255
        def linear(c): return np.where(c<=.04045,c/12.92,((c+.055)/1.055)**2.4)
        blend=linear(base)+(linear(grain)-linear(base))*(.15*f[:,None])
        srgb=np.where(blend<=.0031308,12.92*blend,1.055*blend**(1/2.4)-.055)
        pixels[mask]=np.rint(np.clip(srgb,0,1)*255).astype(np.uint8)
    # Outline visible part boundaries only; never show hidden intersections.
    boundary_owner=np.where(owners>=7,7,owners)
    edge=np.zeros(shape,dtype=bool)
    for n in (1,2):
        edge[n:] |= boundary_owner[n:] != boundary_owner[:-n]; edge[:-n] |= boundary_owner[:-n] != boundary_owner[n:]
        edge[:,n:] |= boundary_owner[:,n:] != boundary_owner[:,:-n]; edge[:,:-n] |= boundary_owner[:,:-n] != boundary_owner[:,n:]
    pixels[edge&(owners>=0)]=rgb(INK)
    paper.paste(Image.fromarray(pixels),(x0*SS,y0*SS))
    yy,xx=np.where(owners>=0)
    actual=[x0+xx.min()/SS,y0+yy.min()/SS,x0+(xx.max()+1)/SS,y0+(yy.max()+1)/SS]
    visible=[int(i) for i in np.unique(owners) if i>=0]
    if name=='TOP': assert all(i in visible for i in range(7)), visible
    expected=([BENCH['nominalSize'][0],BENCH['nominalSize'][2]] if name in ('FRONT','BACK') else
              [BENCH['nominalSize'][1],BENCH['nominalSize'][2]] if name=='RIGHT' else
              [BENCH['nominalSize'][0],BENCH['nominalSize'][1]] if name=='TOP' else None)
    if expected:
        measured=[actual[2]-actual[0],actual[3]-actual[1]]
        assert np.allclose(measured,np.array(expected)*SCALE,atol=1), (name,measured,expected)
    return dict(view=name,cameraDirection=eye.tolist(),screenRight=right.tolist(),
                originPixels=origin,rasterBoundsPixels=actual,visibleGroups=visible)

def dim(a,b,label,vertical=False):
    line([a,b])
    for x,y in (a,b):
        line([(x-7,y+7),(x+7,y-7)])
    x,y=(np.array(a)+b)/2
    text(x-18 if vertical else x,y if vertical else y-35,label,27,anchor='rm' if vertical else 'mt',colour=BLUE)

# Verify documented contact witnesses in analytic default geometry.
def primitive_distance(point, primitive):
    q = np.array(point)+SHIFT-primitive['centre']
    if primitive['kind']=='sphere': return np.linalg.norm(q)-primitive['radius']
    if primitive['kind']=='box':
        d=np.abs(q@primitive['rotation'])-primitive['half']
    else:
        along=np.dot(q,primitive['axis'])
        d=np.array([np.linalg.norm(q-along*primitive['axis'])-primitive['radius'],abs(along)-primitive['half']])
    return np.linalg.norm(np.maximum(d,0))+min(float(np.max(d)),0)


def group_distance(point,group):
    return min(primitive_distance(point,p) for p in PRIMITIVES if p['group']==group)
